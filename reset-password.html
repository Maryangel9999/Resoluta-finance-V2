document.getElementById('sendEmailRecoveryBtn').addEventListener('click', async () => { const emailInput = document.getElementById('recoveryEmail'); const email = emailInput ? emailInput.value.trim() : ''; if (!email) { alert('Por favor, informe seu e-mail.'); return; } if (!window.supabaseClient) { alert('Erro crítico: Cliente Supabase não inicializado.'); return; } const btn = document.getElementById('sendEmailRecoveryBtn'); btn.disabled = true; btn.textContent = 'Enviando...'; try { const { error } = await window.supabaseClient.auth.resetPasswordForEmail(email, { redirectTo: 'https://maryangel9999.github.io/Resoluta-finance-V2/reset-password.html' }); if (error) { throw error; } alert('Link de recuperação enviado com sucesso! Verifique sua caixa de entrada.'); emailInput.value = ''; } catch (err) { alert(err.message || 'Ocorreu um erro ao enviar o e-mail de recuperação.'); } finally { btn.disabled = false; btn.textContent = 'Enviar link de recuperação'; } });
